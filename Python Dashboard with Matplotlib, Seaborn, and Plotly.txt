import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import plotly.graph_objects as go
from plotly.subplots import make_subplots
import warnings
warnings.filterwarnings('ignore')

# Set style
sns.set_style("whitegrid")
plt.rcParams['figure.facecolor'] = 'white'

# ============================================================================
# GENERATE SAMPLE DATA (replace with your own data)
# ============================================================================

np.random.seed(42)
dates = pd.date_range('2024-01-01', periods=365, freq='D')

# Sales data
df_sales = pd.DataFrame({
    'date': dates,
    'revenue': np.cumsum(np.random.randn(365) * 1000 + 5000),
    'orders': np.random.poisson(100, 365),
    'region': np.random.choice(['North', 'South', 'East', 'West'], 365)
})
df_sales['revenue'] = df_sales['revenue'].abs()

# Product categories
categories = ['Electronics', 'Clothing', 'Food', 'Home', 'Sports']
df_products = pd.DataFrame({
    'category': categories,
    'sales': np.random.randint(50000, 200000, len(categories)),
    'profit_margin': np.random.uniform(0.1, 0.4, len(categories))
})

# Customer data
df_customers = pd.DataFrame({
    'segment': ['Enterprise', 'SMB', 'Individual'],
    'count': [150, 450, 2400],
    'avg_value': [15000, 5000, 500]
})

# ============================================================================
# APPROACH 1: STATIC DASHBOARD WITH MATPLOTLIB/SEABORN
# ============================================================================

def create_matplotlib_dashboard():
    """Create a professional dashboard using matplotlib and seaborn"""
    
    fig = plt.figure(figsize=(16, 10))
    fig.suptitle('Sales Performance Dashboard', fontsize=20, fontweight='bold', y=0.98)
    
    # Create grid layout
    gs = fig.add_gridspec(3, 3, hspace=0.3, wspace=0.3)
    
    # 1. Revenue Trend (top row, spans 2 columns)
    ax1 = fig.add_subplot(gs[0, :2])
    df_monthly = df_sales.groupby(df_sales['date'].dt.to_period('M'))['revenue'].sum()
    ax1.plot(df_monthly.index.astype(str), df_monthly.values, linewidth=2.5, color='#2E86AB')
    ax1.fill_between(range(len(df_monthly)), df_monthly.values, alpha=0.3, color='#2E86AB')
    ax1.set_title('Monthly Revenue Trend', fontsize=14, fontweight='bold', pad=10)
    ax1.set_xlabel('Month', fontsize=11)
    ax1.set_ylabel('Revenue ($)', fontsize=11)
    ax1.grid(True, alpha=0.3)
    ax1.tick_params(axis='x', rotation=45)
    
    # 2. KPI Cards (top right)
    ax2 = fig.add_subplot(gs[0, 2])
    ax2.axis('off')
    
    total_revenue = df_sales['revenue'].sum()
    total_orders = df_sales['orders'].sum()
    avg_order = total_revenue / total_orders
    
    kpi_text = f"""
    KEY METRICS
    
    Total Revenue
    ${total_revenue:,.0f}
    
    Total Orders
    {total_orders:,}
    
    Avg Order Value
    ${avg_order:,.2f}
    """
    ax2.text(0.1, 0.5, kpi_text, fontsize=11, verticalalignment='center',
             bbox=dict(boxstyle='round', facecolor='lightblue', alpha=0.3))
    
    # 3. Sales by Category (middle left)
    ax3 = fig.add_subplot(gs[1, 0])
    colors = sns.color_palette('Set2', len(df_products))
    bars = ax3.barh(df_products['category'], df_products['sales'], color=colors)
    ax3.set_title('Sales by Category', fontsize=14, fontweight='bold', pad=10)
    ax3.set_xlabel('Sales ($)', fontsize=11)
    
    # Add value labels
    for bar in bars:
        width = bar.get_width()
        ax3.text(width, bar.get_y() + bar.get_height()/2, 
                f'${width:,.0f}', ha='left', va='center', fontsize=9)
    
    # 4. Profit Margin Analysis (middle center)
    ax4 = fig.add_subplot(gs[1, 1])
    ax4.scatter(df_products['sales'], df_products['profit_margin'] * 100, 
               s=300, c=colors, alpha=0.6, edgecolors='black', linewidth=1.5)
    
    for i, cat in enumerate(df_products['category']):
        ax4.annotate(cat, 
                    (df_products['sales'].iloc[i], df_products['profit_margin'].iloc[i] * 100),
                    fontsize=9, ha='center')
    
    ax4.set_title('Sales vs Profit Margin', fontsize=14, fontweight='bold', pad=10)
    ax4.set_xlabel('Sales ($)', fontsize=11)
    ax4.set_ylabel('Profit Margin (%)', fontsize=11)
    ax4.grid(True, alpha=0.3)
    
    # 5. Regional Distribution (middle right)
    ax5 = fig.add_subplot(gs[1, 2])
    region_sales = df_sales.groupby('region')['revenue'].sum()
    wedges, texts, autotexts = ax5.pie(region_sales.values, labels=region_sales.index,
                                        autopct='%1.1f%%', startangle=90,
                                        colors=sns.color_palette('pastel'))
    ax5.set_title('Revenue by Region', fontsize=14, fontweight='bold', pad=10)
    
    for autotext in autotexts:
        autotext.set_color('black')
        autotext.set_fontsize(10)
        autotext.set_fontweight('bold')
    
    # 6. Customer Segments (bottom left)
    ax6 = fig.add_subplot(gs[2, 0])
    ax6.bar(df_customers['segment'], df_customers['count'], color='#A23B72', alpha=0.7)
    ax6.set_title('Customer Count by Segment', fontsize=14, fontweight='bold', pad=10)
    ax6.set_ylabel('Number of Customers', fontsize=11)
    ax6.tick_params(axis='x', rotation=45)
    
    # 7. Orders Over Time (bottom center and right)
    ax7 = fig.add_subplot(gs[2, 1:])
    df_weekly = df_sales.groupby(df_sales['date'].dt.to_period('W'))['orders'].sum()
    ax7.plot(df_weekly.index.astype(str), df_weekly.values, color='#F18F01', linewidth=2)
    ax7.set_title('Weekly Orders Trend', fontsize=14, fontweight='bold', pad=10)
    ax7.set_xlabel('Week', fontsize=11)
    ax7.set_ylabel('Number of Orders', fontsize=11)
    ax7.grid(True, alpha=0.3)
    ax7.tick_params(axis='x', rotation=45)
    
    # Only show every 4th week label to avoid crowding
    tick_positions = range(0, len(df_weekly), 4)
    ax7.set_xticks(tick_positions)
    ax7.set_xticklabels([df_weekly.index.astype(str)[i] for i in tick_positions])
    
    plt.tight_layout()
    return fig

# APPROACH 2: INTERACTIVE DASHBOARD WITH PLOTLY
# ============================================================================

def create_plotly_dashboard():
    """Create an interactive dashboard using Plotly"""
    
    # Create subplots
    fig = make_subplots(
        rows=3, cols=3,
        subplot_titles=('Monthly Revenue Trend', 'Sales by Category', 'Profit Margin Analysis',
                       'Regional Distribution', 'Customer Segments', 'Weekly Orders Trend'),
        specs=[
            [{'type': 'scatter', 'colspan': 2}, None, {'type': 'indicator'}],
            [{'type': 'bar'}, {'type': 'scatter'}, {'type': 'pie'}],
            [{'type': 'bar'}, {'type': 'scatter', 'colspan': 2}, None]
        ],
        vertical_spacing=0.12,
        horizontal_spacing=0.1
    )
    
    # 1. Revenue Trend
    df_monthly = df_sales.groupby(df_sales['date'].dt.to_period('M'))['revenue'].sum()
    fig.add_trace(
        go.Scatter(x=df_monthly.index.astype(str), y=df_monthly.values,
                  mode='lines+markers', name='Revenue',
                  line=dict(color='#2E86AB', width=3),
                  fill='tozeroy', fillcolor='rgba(46, 134, 171, 0.2)'),
        row=1, col=1
    )
    
    # 2. KPI Indicator
    total_revenue = df_sales['revenue'].sum()
    fig.add_trace(
        go.Indicator(
            mode='number+delta',
            value=total_revenue,
            title={'text': 'Total Revenue', 'font': {'size': 18}},
            number={'prefix': '$', 'font': {'size': 30}},
            delta={'reference': total_revenue * 0.9, 'relative': True}
        ),
        row=1, col=3
    )
    
    # 3. Sales by Category
    fig.add_trace(
        go.Bar(x=df_products['sales'], y=df_products['category'],
              orientation='h', name='Sales',
              marker=dict(color=df_products['sales'], 
                         colorscale='Viridis')),
        row=2, col=1
    )
    
    # 4. Profit Margin Scatter
    fig.add_trace(
        go.Scatter(x=df_products['sales'], 
                  y=df_products['profit_margin'] * 100,
                  mode='markers+text',
                  text=df_products['category'],
                  textposition='top center',
                  marker=dict(size=15, color=df_products['sales'],
                            colorscale='Portland', showscale=True),
                  name='Products'),
        row=2, col=2
    )
    
    # 5. Regional Distribution
    region_sales = df_sales.groupby('region')['revenue'].sum()
    fig.add_trace(
        go.Pie(labels=region_sales.index, values=region_sales.values,
              hole=0.3, name='Regions'),
        row=2, col=3
    )
    
    # 6. Customer Segments
    fig.add_trace(
        go.Bar(x=df_customers['segment'], y=df_customers['count'],
              name='Customers',
              marker=dict(color=['#A23B72', '#F18F01', '#C73E1D'])),
        row=3, col=1
    )
    
    # 7. Weekly Orders
    df_weekly = df_sales.groupby(df_sales['date'].dt.to_period('W'))['orders'].sum()
    fig.add_trace(
        go.Scatter(x=df_weekly.index.astype(str), y=df_weekly.values,
                  mode='lines', name='Orders',
                  line=dict(color='#F18F01', width=2)),
        row=3, col=2
    )
    
    # Update layout
    fig.update_layout(
        title_text='Sales Performance Dashboard (Interactive)',
        title_font_size=24,
        showlegend=False,
        height=1000,
        template='plotly_white'
    )
    
    # Update axes labels
    fig.update_xaxes(title_text='Month', row=1, col=1)
    fig.update_yaxes(title_text='Revenue ($)', row=1, col=1)
    fig.update_xaxes(title_text='Sales ($)', row=2, col=1)
    fig.update_xaxes(title_text='Sales ($)', row=2, col=2)
    fig.update_yaxes(title_text='Profit Margin (%)', row=2, col=2)
    fig.update_xaxes(title_text='Segment', row=3, col=1)
    fig.update_yaxes(title_text='Count', row=3, col=1)
    fig.update_xaxes(title_text='Week', row=3, col=2)
    fig.update_yaxes(title_text='Orders', row=3, col=2)
    
    return fig
